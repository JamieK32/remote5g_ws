cmake_minimum_required(VERSION 3.16)
project(remote5g_sim)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(gazebo REQUIRED)
foreach(dep gazebo_ros rclcpp ackermann_msgs geometry_msgs nav_msgs sensor_msgs std_msgs std_srvs diagnostic_msgs tf2_ros)
  find_package(${dep} REQUIRED)
endforeach()
find_package(Qt5Widgets REQUIRED)
add_library(remote5g_gui_capture SHARED src/gui_capture.cpp)
target_include_directories(remote5g_gui_capture PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(remote5g_gui_capture PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(remote5g_gui_capture ${GAZEBO_LIBRARIES} Qt5::Widgets)
install(TARGETS remote5g_gui_capture LIBRARY DESTINATION lib)
add_library(remote5g_vehicle SHARED src/vehicle_plugin.cpp)
target_include_directories(remote5g_vehicle PRIVATE include ${GAZEBO_INCLUDE_DIRS})
target_link_directories(remote5g_vehicle PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(remote5g_vehicle ${GAZEBO_LIBRARIES})
ament_target_dependencies(remote5g_vehicle gazebo_ros rclcpp ackermann_msgs geometry_msgs nav_msgs sensor_msgs std_msgs std_srvs diagnostic_msgs tf2_ros)
target_compile_options(remote5g_vehicle PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS remote5g_vehicle LIBRARY DESTINATION lib)
install(DIRECTORY urdf config launch worlds docs DESTINATION share/${PROJECT_NAME})
# Both launchers use installed track data, independent of the source checkout.
install(DIRECTORY ../../config ../../simulation DESTINATION share/${PROJECT_NAME}/track)
install(FILES README.md LICENSE DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/drive.py scripts/inspect_cameras.py test/regression.py test/model_check.py test/smoke.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(control_math_test test/control_math_test.cpp)
  target_include_directories(control_math_test PRIVATE include)
  add_test(NAME control_math COMMAND control_math_test)
  set_tests_properties(control_math PROPERTIES TIMEOUT 10)
endif()
ament_package()
