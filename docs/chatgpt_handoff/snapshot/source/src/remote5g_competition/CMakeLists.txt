cmake_minimum_required(VERSION 3.16)
project(remote5g_competition)
find_package(ament_cmake REQUIRED)
find_package(gazebo REQUIRED)
find_package(Qt5Widgets REQUIRED)
set(CMAKE_CXX_STANDARD 17)
add_library(competition_capture SHARED src/gui_capture.cpp)
target_include_directories(competition_capture PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_directories(competition_capture PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(competition_capture ${GAZEBO_LIBRARIES} Qt5::Widgets)
install(TARGETS competition_capture LIBRARY DESTINATION lib)
install(DIRECTORY models config launch worlds materials docs scripts DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/generate_layout.py scripts/validate_layout.py scripts/manage_props.py scripts/parking_check.py test/integration.py test/run_matrix.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/scene_core.py DESTINATION lib/${PROJECT_NAME})
install(FILES README.md LICENSE DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 enable_testing()
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME geometry COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_geometry.py)
 set_tests_properties(geometry PROPERTIES TIMEOUT 45)
endif()
ament_package()
